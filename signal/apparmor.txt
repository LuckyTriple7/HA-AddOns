#include <tunables/global>

profile signal_addon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  capability chown,
  capability dac_override,
  capability fowner,
  capability setgid,
  capability setuid,
  capability sys_admin,
  file,
  signal,
  network tcp,
  network udp,
  network inet,
  network inet6,
  network netlink raw,
  network unix dgram,

  # Debian-Basis (bbernhard/signal-cli-rest-api): ENTRYPOINT /run.sh startet
  # /entrypoint.sh, signal-cli (Java) liegt unter /opt, die JRE unter /usr/lib.
  /run.sh ix,
  /entrypoint.sh ix,
  /run/{,**} rwk,
  /dev/tty rw,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /usr/lib/** ix,
  /lib/** ix,
  /opt/** ix,
  /sbin/** ix,
  /usr/sbin/** ix,
  /var/run/** mrwkl,
  /var/run/ mrwkl,
  /dev/* mrwkl,
  /tmp/** mrkwl,

  /ui/** rwix,
  /data/** rw,
  /config/** rw,

  ptrace (trace,read) peer=docker-default,
  signal (receive) set=(kill,term) peer=/usr/bin/docker,
}
